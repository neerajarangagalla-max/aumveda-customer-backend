import logging

from fastapi import APIRouter, HTTPException, status

from app.core.supabase import supabase


router = APIRouter(
    prefix="/home",
    tags=["Home"],
)

logger = logging.getLogger(__name__)


@router.get("/hero")
async def get_home_hero():
    try:
        response = (
            supabase
            .table("hero_banners")
            .select(
    "id,"
    "eyebrow,"
    "title_line1,"
    "title_line2,"
    "description,"
    "image_url,"
    "video_url,"
    "primary_cta_text,"
    "primary_cta_path,"
    "secondary_cta_text,"
    "secondary_cta_path"
)
            .eq("is_active", True)
            .order("sort_order")
            .limit(1)
            .execute()
        )

        if not response.data:
            raise HTTPException(
                status_code=status.HTTP_404_NOT_FOUND,
                detail="Active hero banner not found",
            )

        return {
            "status": "success",
            "data": response.data[0],
        }

    except HTTPException:
        raise

    except Exception:
        logger.exception("Failed to retrieve homepage hero")

        raise HTTPException(
            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
            detail="Unable to retrieve homepage hero",
        )

@router.get("/featured-products")
async def get_featured_products():

    try:

        response = (
            supabase
            .table("products")
            .select(
                """
                id,
                name,
                slug,
                short_description,
                average_rating,
                review_count,

                product_variants(
                    id,
                    name,
                    price,
                    compare_at_price,
                    stock_quantity,
                    is_default
                ),

                product_images(
                    id,
                    image_url,
                    alt_text,
                    is_primary,
                    sort_order
                )
                """
            )
            .eq(
                "is_featured",
                True
            )
            .eq(
                "status",
                "active"
            )
            .order(
                "created_at",
                desc=False
            )
            .execute()
        )


        products = []


        for product in response.data:

            variants = (
                product.get(
                    "product_variants",
                    []
                )
            )


            images = (
                product.get(
                    "product_images",
                    []
                )
            )


            default_variant = None

            for variant in variants:

                if variant.get(
                    "is_default"
                ):
                    default_variant = variant
                    break


            primary_image = None

            for image in images:

                if image.get(
                    "is_primary"
                ):
                    primary_image = image
                    break



            products.append(
                {
                    "id": product["id"],

                    "name": product["name"],

                    "slug": product["slug"],

                    "description":
                        product.get(
                            "short_description"
                        ),

                    "rating":
                        product.get(
                            "average_rating",
                            0
                        ),

                    "review_count":
                        product.get(
                            "review_count",
                            0
                        ),

                    "price":
                        default_variant.get(
                            "price"
                        )
                        if default_variant
                        else None,


                    "compare_at_price":
                        default_variant.get(
                            "compare_at_price"
                        )
                        if default_variant
                        else None,


                    "variant":
                        default_variant,


                    "image":
                        primary_image.get(
                            "image_url"
                        )
                        if primary_image
                        else None
                }
            )



        return {

            "status": "success",

            "count": len(products),

            "data": products

        }


    except Exception as error:

        logger.exception(
            "Featured products API failed"
        )

        raise HTTPException(
            status_code=500,
            detail=str(error)
        )

from fastapi import APIRouter, HTTPException

from app.core.supabase import supabase


router = APIRouter(
    prefix="/products",
    tags=["Products"]
)


# ============================================================
# GET ALL ACTIVE PRODUCTS
# ============================================================

@router.get("")
def get_products():

    try:

        response = (
            supabase
            .table("products")
            .select(
                """
                id,
                vendor_id,
                category_id,
                name,
                slug,
                short_description,
                description,
                status,
                is_featured,
                average_rating,
                review_count,
                product_images(
                    image_url,
                    alt_text,
                    sort_order,
                    is_primary
                ),
                product_variants(
                    id,
                    name,
                    price,
                    compare_at_price,
                    stock_quantity,
                    is_default,
                    is_active
                )
                """
            )
            .eq("status", "active")
            .execute()
        )

        products = []

        for product in response.data:

            # ------------------------------------------------
            # PRIMARY IMAGE
            # ------------------------------------------------

            images = product.get("product_images") or []

            images = sorted(
                images,
                key=lambda image: (
                    not image.get("is_primary", False),
                    image.get("sort_order") or 9999
                )
            )

            image_url = None

            if images:
                image_url = images[0].get("image_url")


            # ------------------------------------------------
            # DEFAULT ACTIVE VARIANT
            # ------------------------------------------------

            variants = product.get("product_variants") or []

            active_variants = [
                variant
                for variant in variants
                if variant.get("is_active", False)
            ]

            default_variant = next(
                (
                    variant
                    for variant in active_variants
                    if variant.get("is_default", False)
                ),
                None
            )

            if default_variant is None and active_variants:
                default_variant = active_variants[0]


            price = None
            compare_at_price = None
            stock_quantity = 0
            variant_id = None
            variant_name = None


            if default_variant:

                variant_id = default_variant.get("id")

                variant_name = default_variant.get("name")

                price = default_variant.get("price")

                compare_at_price = default_variant.get(
                    "compare_at_price"
                )

                stock_quantity = (
                    default_variant.get("stock_quantity") or 0
                )


            # ------------------------------------------------
            # FINAL FRONTEND RESPONSE
            # ------------------------------------------------

            products.append({

                "id": product.get("id"),

                "vendor_id": product.get("vendor_id"),

                "category_id": product.get("category_id"),

                "name": product.get("name"),

                "slug": product.get("slug"),

                "short_description":
                    product.get("short_description"),

                "description":
                    product.get("description"),

                "image": image_url,

                "price": price,

                "compare_at_price":
                    compare_at_price,

                "rating":
                    product.get("average_rating"),

                "review_count":
                    product.get("review_count"),

                "is_featured":
                    product.get("is_featured", False),

                "variant_id":
                    variant_id,

                "variant_name":
                    variant_name,

                "stock_quantity":
                    stock_quantity,

                "in_stock":
                    stock_quantity > 0

            })


        return products


    except Exception as error:

        raise HTTPException(
            status_code=500,
            detail=str(error)
        )

@router.get("/{slug}")
def get_product_by_slug(slug: str):

    try:

        response = (
            supabase
            .table("products")
            .select(
                """
                *,
                product_images(*),
                product_variants(*)
                """
            )
            .eq(
                "slug",
                slug
            )
            .single()
            .execute()
        )

        if not response.data:

            raise HTTPException(
                status_code=404,
                detail="Product not found"
            )

        return response.data


    except HTTPException:

        raise


    except Exception:

        raise HTTPException(
            status_code=404,
            detail="Product not found"
        )

from functools import lru_cache

from supabase import Client, create_client

from app.core.config import settings


@lru_cache
def get_supabase_client() -> Client:
    client: Client = create_client(
        settings.supabase_url,
        settings.supabase_secret_key,
    )

    return client


supabase: Client = get_supabase_client()
